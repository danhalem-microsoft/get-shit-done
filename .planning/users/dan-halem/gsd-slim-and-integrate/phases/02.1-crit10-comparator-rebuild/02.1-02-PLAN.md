class OpenAIKeyUnsetError extends Error {
  constructor(msg) { super(msg); this.name = 'OpenAIKeyUnsetError'; }

const EMBEDDINGS_CACHE_DIR = path.resolve(__dirname, '..', 'test-fixtures', 'baselines', 'embeddings');

 * @param {string[]} titles - raw titles (will be normalized inside)
 * @param {object} opts - { model, apiKey, cacheBucket, timeoutMs }
 * @returns {Promise<Map<string, number[]>>} - keyed by NORMALIZED title -> vector

async function embedTitles(titles, opts = {}) {
  const apiKey = opts.apiKey ?? process.env.OPENAI_API_KEY;
  if (!apiKey) throw new OpenAIKeyUnsetError('OPENAI_API_KEY must be set');

       function cosineSimilarity(vecA, vecB) {
         if (!Array.isArray(vecA) || !Array.isArray(vecB)) {
           throw new Error('cosineSimilarity: both inputs must be number arrays');
         }
         if (vecA.length !== vecB.length) {
           throw new Error(`cosineSimilarity: length mismatch ${vecA.length} vs ${vecB.length}`);
         }
         let dot = 0, normA = 0, normB = 0;
         for (let i = 0; i < vecA.length; i++) {
           dot   += vecA[i] * vecB[i];
           normA += vecA[i] * vecA[i];
           normB += vecB[i] * vecB[i];
         }
         if (normA === 0 || normB === 0) return 0;  // zero-vector defense
         return dot / (Math.sqrt(normA) * Math.sqrt(normB));
       }
