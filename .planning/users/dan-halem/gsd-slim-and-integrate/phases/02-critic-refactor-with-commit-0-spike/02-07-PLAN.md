```javascript
'use strict';

const { test, describe } = require('node:test');
const assert = require('node:assert');
const fs = require('node:fs');
const path = require('node:path');
const { runClaudeWithTools, getRepoRoot } = require('./helpers/claude-runner.cjs');
const { recordWalltime } = require('./helpers/walltime-recorder.cjs');

// CRIT-09: 1-of-N failure injection — orchestrator must skip-and-continue.
// RESEARCH §Open-Q-3 Option-a: bad subagent_type.

const FIXTURE_PHASE = '1';

describe('CRIT-09: orchestrator skip-and-continue when 1-of-N critic fails', () => {
  test('orchestrator continues with 5 critics + logs missing critic as info severity', async () => {
    const result = await runClaudeWithTools(
      `Run /gsd-review --critique ${FIXTURE_PHASE} --inject-fault discuss`,
      { cwd: getRepoRoot(), timeout: 600_000, maxBudget: 30 }
    );

    recordWalltime({
      test: 'integration:critic-fault-injection',
      walltime_ms: result.duration_ms,
      cost_usd: result.cost,
      phase: 'phase-2-critic',
    });

    assert.ok(result.success, `orchestrator must continue when 1 critic errors; got: ${result.error}`);

    // Read merged CRITIQUE.md from the phase dir
    const phaseDir = locatePhaseDir(getRepoRoot(), FIXTURE_PHASE);
    const merged = fs.readFileSync(path.join(phaseDir, 'CRITIQUE.md'), 'utf-8');

    // Surviving 5 critics aggregated
    for (const lens of ['plan', 'code', 'scope', 'verify', 'strategy']) {
      assert.match(merged, new RegExp(`critique_type:\\s*${lens}|CRITIQUE-${lens}`),
        `merged CRITIQUE.md must reference surviving critic ${lens}`);
    }

    // Failed critic logged as info finding
    assert.match(merged, /\[INFO\][\s\S]*discuss[\s\S]*did not produce CRITIQUE/i,
      'orchestrator must log missing critic as info-severity finding (per workflows/critique.md Step 5)');
    assert.match(merged, /missing-critic-output/i,
      'orchestrator finding must use category missing-critic-output');
  });
});

function locatePhaseDir(repoRoot, phaseArg) {
  const { execFileSync } = require('node:child_process');
  const out = execFileSync('node', [
    path.join(repoRoot, 'get-shit-done', 'bin', 'gsd-tools.cjs'),
    'find-phase', phaseArg
  ], { cwd: repoRoot, encoding: 'utf-8' });
  return out.trim();
}
```
