const { test, describe } = require('node:test');

const fs = require('node:fs');
const path = require('node:path');

const ROOT = path.resolve(__dirname, '..');
const {
  survivingCommandCount,
  survivingAgentCount,
  deprecationStubs,

// --- INVENTORY.md parser ---------------------------------------------------
// INVENTORY.md groups commands and agents under headers like "## Commands (N shipped)"
// and lists each entry as a markdown row (e.g., a table row or bullet referencing
// commands/gsd/<name>.md or agents/<name>.md). Parse by extracting `<name>.md`
// references inside the corresponding section.
function parseInventory() {
  const inventoryPath = path.join(ROOT, 'docs', 'INVENTORY.md');
  const text = fs.readFileSync(inventoryPath, 'utf-8');

  // Split by top-level headers; find the Commands and Agents sections.
  // Header regex tolerates both "## Commands (N shipped)" and "## Commands".
  const sections = text.split(/^##\s+/m);
  const cmdSection = sections.find((s) => /^Commands\b/i.test(s)) || '';
  const agentSection = sections.find((s) => /^Agents\b/i.test(s)) || '';

  // Extract command names: any reference to commands/gsd/<name>.md in the section.
  const cmdNames = new Set();
  for (const m of cmdSection.matchAll(/commands\/gsd\/([a-z0-9_-]+)\.md/g)) {
    cmdNames.add(m[1]);
  }

  // Extract agent names: any reference to agents/<name>.md in the section.
  const agentNames = new Set();
  for (const m of agentSection.matchAll(/agents\/([a-z0-9_-]+)\.md/g)) {
    agentNames.add(m[1]);
  }

  return { cmdNames, agentNames };
}
